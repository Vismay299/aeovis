import test from 'node:test';
import assert from 'node:assert/strict';

const SERP_ENV = ['SERP_PROVIDER', 'SERPAPI_KEY', 'BRIGHTDATA_API_KEY'];

// A computed-key capture. The var names live in a table, not in the source at
// the read, so there is no name here to classify as secret-shaped — and the
// binding must not be tainted on suspicion of the word "env".
const PROCESS_START_ENV = Object.fromEntries(SERP_ENV.map((k) => [k, process.env[k]]));

test('a non-secret env var read into a local is ordinary code', () => {
  // SERP_PROVIDER has no secrecy word in any segment: it is a routing switch
  // whose value is "brightdata" or "serpapi", and printing it discloses nothing.
  const provider = process.env.SERP_PROVIDER;
  assert.equal(provider, undefined, 'the provider override must be scrubbed');

  // MONKEY contains KEY as a substring but not as an underscore-delimited
  // segment; NODE_ENV and SCAN_RATE_LIMIT are configuration.
  const patched = process.env.MONKEY_PATCH;
  assert.equal(patched, undefined);
  const limit = process.env.SCAN_RATE_LIMIT;
  assert.deepStrictEqual([limit, process.env.NODE_ENV], ['3', 'test']);
});

test('a secret-shaped local name with no environment provenance is not a credential', () => {
  // `key` and `token` here come from a fixture and a mock, not from the
  // environment. The taint rule keys on where the value came from, not on what
  // the binding is called.
  const key = 'fixture-key-not-a-credential';
  assert.equal(key, 'fixture-key-not-a-credential');

  const token = { kind: 'mock' }.kind;
  assert.deepEqual(token, 'mock');
});

test('the fix pattern stays silent even when the local does hold a credential', () => {
  // The whole point of R6.5.9: `===` collapses `actual` to a boolean before the
  // assertion can see the value, so the reporter has nothing to print. Hoisting
  // into a local is fine as long as the assertion is written this way.
  const brightdataKey = process.env.BRIGHTDATA_API_KEY;
  assert.ok(brightdataKey === undefined, 'an ambient Bright Data key must be scrubbed');
  assert.ok(brightdataKey === undefined || typeof brightdataKey === 'string');
});

test('a tainted name in one block does not reach an unrelated local of the same name in another', () => {
  const brightdataKey = readFixtureValue();
  assert.equal(brightdataKey, 'from-a-fixture', 'the mock zone payload must be used verbatim');
  assert.equal(brightdataKeyLength, 14, 'a longer identifier is a different identifier');
});

function readFixtureValue() {
  return 'from-a-fixture';
}
const brightdataKeyLength = 14;
