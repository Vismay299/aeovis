import test from 'node:test';
import assert from 'node:assert/strict';

const SERP_ENV = ['SERP_PROVIDER', 'SERPAPI_KEY', 'BRIGHTDATA_API_KEY'];

/** Shape of a value, never the value. */
function describeEnv(v) {
  return v === undefined ? 'unset' : `set (len ${v.length})`;
}

function useSerpEnv(t, env) {
  const saved = Object.fromEntries(SERP_ENV.map((k) => [k, process.env[k]]));
  for (const k of SERP_ENV) delete process.env[k];
  for (const [k, v] of Object.entries(env)) process.env[k] = v;
  t.after(() => {
    for (const k of SERP_ENV) {
      if (saved[k] === undefined) delete process.env[k];
      else process.env[k] = saved[k];
    }
  });
}

test('GUARD — an ambient Bright Data key is scrubbed, not inherited', (t) => {
  useSerpEnv(t, { SERPAPI_KEY: 'mock-key' });

  // The hoist. Reads as a tidy local; is a latent disclosure.
  const brightdataKey = process.env.BRIGHTDATA_API_KEY;
  assert.equal(brightdataKey, undefined, `must be scrubbed; ${describeEnv(brightdataKey)}`); // LEAKS

  // Same defect through bracket access, which is how it gets written when the
  // var name is built from a table.
  const cron = process.env['CRON_SECRET'];
  assert.strictEqual(cron, undefined, 'the cron secret must not be inherited'); // LEAKS

  // Same defect through destructuring, the other one-line spelling.
  const { SUPABASE_SERVICE_ROLE_KEY } = process.env;
  assert.deepEqual([SUPABASE_SERVICE_ROLE_KEY], [undefined], 'service role key must be scrubbed'); // LEAKS

  // Correct: the comparison collapses `actual` to a boolean before the assertion
  // can see the value, so there is nothing left for the reporter to print.
  assert.ok(process.env.SERPAPI_KEY === 'mock-key', `mock key must apply; ${describeEnv(process.env.SERPAPI_KEY)}`);

  // Correct: a non-credential var is not this guard's business.
  assert.equal(process.env.SERP_PROVIDER, undefined, 'the provider override must be scrubbed');
});
