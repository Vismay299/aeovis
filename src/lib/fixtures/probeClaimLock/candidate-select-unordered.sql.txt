-- FIXTURE — R1.2.15. The lock's ORDERING semantics broken: LIMIT over an
-- unordered candidate set, with the clause itself still present and in place.
--
-- R1215-EXPECT: reject /no ORDER BY/
-- R1215-DEFEATS: whole-file-regex, comment-stripped-regex, single-executable-clause
-- R1215-DERIVED-FROM: base-as-in-migration-010.sql.txt
--
-- WHERE IT CAME FROM. base-as-in-migration-010.sql.txt with ONE change: the
--
--     ORDER BY t.priority DESC, t.created_at
--
-- line deleted from the candidate SELECT. `LIMIT p_limit FOR UPDATE SKIP LOCKED`
-- still closes the SELECT in that order, so the lock is present, executable,
-- singular, correctly scoped and last.
--
-- WHAT MAKES IT INTERESTING. It is valid SQL — which the other ordering attacks
-- are not. Postgres requires the locking clause to come LAST, so a fixture with
-- `FOR UPDATE SKIP LOCKED` placed ahead of ORDER BY/LIMIT would be a syntax
-- error and could never reach production; this one applies cleanly and runs. It
-- is also invisible to every lock-shaped check, and it is the one sabotage in
-- this directory whose damage is a DISTRIBUTION rather than a hard error, so a
-- test that only counted claims would report perfect health.
--
-- WHY IT IS WRONG. `LIMIT n` without ORDER BY takes whichever n rows the plan
-- happens to produce, and with FOR UPDATE SKIP LOCKED under contention that is
-- whichever rows happen not to be locked. Two things follow. First, priority
-- stops meaning anything: the `probe_tasks_claim_idx` index exists precisely to
-- serve `(engine, priority DESC, created_at)`, and a claim that does not ask for
-- that order will not get it. Second, a task can starve indefinitely — an
-- unlucky row is skipped every poll, its lease never taken, and because no probe
-- ever fails there is nothing to report. The scan finishes short and calls it
-- done, which is the false success this rebuild exists to remove.
--
-- FOR UPDATE SKIP LOCKED is ordering-dependent by nature: "skip the locked ones
-- and take the next" is only a fair queue if "next" is defined. This fixture is
-- the reminder that the clause's semantics are not carried by the clause alone.
--
-- This file is .txt on purpose: it must never be applied as a migration, and
-- the migration walker in probeClaimLock.test.mjs must never find it.

CREATE OR REPLACE FUNCTION claim_probe_tasks(
  p_owner     text,
  p_engines   text[]  DEFAULT NULL,
  p_limit     integer DEFAULT 5,
  p_lease_ms  integer DEFAULT 180000,
  p_batch_id  uuid    DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
AS $fn$
DECLARE
  v_now     timestamptz := now();
  v_expires timestamptz;
  v_claimed jsonb;
  v_count   integer;
BEGIN
  v_expires := v_now + make_interval(secs => p_lease_ms / 1000.0);

  WITH candidate AS (
    SELECT t.id
      FROM probe_tasks t
     WHERE t.status IN ('pending', 'claimed')
       AND (p_engines  IS NULL OR t.engine = ANY (p_engines))
       AND (p_batch_id IS NULL OR t.batch_id = p_batch_id)
       AND (t.lease_owner IS NULL
            OR t.lease_expires_at IS NULL
            OR t.lease_expires_at < v_now)
     LIMIT p_limit
     FOR UPDATE SKIP LOCKED
  ),
  claimed AS (
    UPDATE probe_tasks t
       SET status           = 'claimed',
           lease_owner      = p_owner,
           lease_expires_at = v_expires,
           attempts         = t.attempts + 1,
           claimed_at       = v_now,
           updated_at       = v_now
      FROM candidate c
     WHERE t.id = c.id
    RETURNING t.*
  )
  SELECT COALESCE(jsonb_agg(to_jsonb(claimed) ORDER BY claimed.created_at), '[]'::jsonb),
         COUNT(*)::integer
    INTO v_claimed, v_count
    FROM claimed;

  RETURN jsonb_build_object(
    'owner',         p_owner,
    'tasks',         v_claimed,
    'claimed_count', v_count
  );
END;
$fn$;
