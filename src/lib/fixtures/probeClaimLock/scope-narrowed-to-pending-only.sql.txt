-- FIXTURE — R1.2.15. The claim window NARROWED: a task whose worker died can
-- never be reclaimed, and the lock is untouched.
--
-- R1215-EXPECT: reject /LAPSED/
-- R1215-DEFEATS: whole-file-regex, comment-stripped-regex, single-executable-clause
-- R1215-DERIVED-FROM: base-as-in-migration-010.sql.txt
--
-- WHERE IT CAME FROM. base-as-in-migration-010.sql.txt with ONE change:
--
--     WHERE t.status IN ('pending', 'claimed')   ->   WHERE t.status = 'pending'
--
-- This is not an invented attack. Migration 010's own comment records that its
-- FIRST DRAFT shipped exactly this, and that the live lease-expiry test caught
-- it. This fixture is that regression, committed, so it stays caught in an
-- environment that has no database to catch it with.
--
-- WHAT MAKES IT INTERESTING. Like the widening fixture, it leaves the lock
-- alone: whole-file regex, comment-stripped regex, one executable clause, inside
-- the candidate CTE, last clause of the SELECT. It is also the narrowing that
-- LOOKS most defensible to a reader — "claim the pending ones" reads as obviously
-- correct, and the reason it is not lives three CTEs away in the reaper.
--
-- WHY IT IS WRONG. A worker that dies mid-probe leaves its row at 'claimed'
-- with a lease that will lapse in 180s. That row is the ONLY kind of row the
-- reclaim path exists for. Restrict the window to 'pending' and LEASE_TTL_MS
-- means nothing: the lease expires and no worker can ever pick the task up
-- again. The failure is silent in the worst way — the queue drains, every
-- remaining task is invisible to every claimant, and the scan reports fewer
-- probes rather than an error. A stranded task is not a failed task; it is a
-- missing one, and missing is what standing rule 1 bans.
--
-- Asserted behaviourally, not textually: the suite runs this fixture's own SQL
-- and shows the lapsed-lease row admitted by NO claimant. See "TEETH —
-- narrowing the window strands a task whose worker died" in
-- src/lib/probeClaimLock.test.mjs.
--
-- This file is .txt on purpose: it must never be applied as a migration, and
-- the migration walker in probeClaimLock.test.mjs must never find it.

CREATE OR REPLACE FUNCTION claim_probe_tasks(
  p_owner     text,
  p_engines   text[]  DEFAULT NULL,
  p_limit     integer DEFAULT 5,
  p_lease_ms  integer DEFAULT 180000,
  p_batch_id  uuid    DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
AS $fn$
DECLARE
  v_now     timestamptz := now();
  v_expires timestamptz;
  v_claimed jsonb;
  v_count   integer;
BEGIN
  v_expires := v_now + make_interval(secs => p_lease_ms / 1000.0);

  WITH candidate AS (
    SELECT t.id
      FROM probe_tasks t
     WHERE t.status = 'pending'
       AND (p_engines  IS NULL OR t.engine = ANY (p_engines))
       AND (p_batch_id IS NULL OR t.batch_id = p_batch_id)
       AND (t.lease_owner IS NULL
            OR t.lease_expires_at IS NULL
            OR t.lease_expires_at < v_now)
     ORDER BY t.priority DESC, t.created_at
     LIMIT p_limit
     FOR UPDATE SKIP LOCKED
  ),
  claimed AS (
    UPDATE probe_tasks t
       SET status           = 'claimed',
           lease_owner      = p_owner,
           lease_expires_at = v_expires,
           attempts         = t.attempts + 1,
           claimed_at       = v_now,
           updated_at       = v_now
      FROM candidate c
     WHERE t.id = c.id
    RETURNING t.*
  )
  SELECT COALESCE(jsonb_agg(to_jsonb(claimed) ORDER BY claimed.created_at), '[]'::jsonb),
         COUNT(*)::integer
    INTO v_claimed, v_count
    FROM claimed;

  RETURN jsonb_build_object(
    'owner',         p_owner,
    'tasks',         v_claimed,
    'claimed_count', v_count
  );
END;
$fn$;
